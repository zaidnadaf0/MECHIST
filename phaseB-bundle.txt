import * as functions from 'firebase-functions';
import * as admin from 'firebase-admin';
import { ServiceRequest, MechanicProfile } from './types';
import {
  findEligibleMechanics,
  resetRequestForRematch,
  needsRematching,
  OFFER_WINDOW_MS,
  deriveEtaMinutes,
} from './matching';

admin.initializeApp();
const db = admin.firestore();

/**
 * Triggered when a request document is written.
 * Handles:
 *   - Auto-rematch when a mechanic declines (declinedBy array grows)
 *   - Auto-rematch when a mechanic releases (status back to requested, mechanicId cleared)
 *   - Initial dispatch on create (if no pickedMechanicId)
 */
export const onRequestWrite = functions.firestore
  .document('requests/{requestId}')
  .onWrite(async (change, context) => {
    const requestId = context.params.requestId;
    const before = change.before.exists ? (change.before.data() as ServiceRequest) : null;
    const after = change.after.exists ? (change.after.data() as ServiceRequest) : null;

    if (!after) {
      console.log(`[onRequestWrite] ${requestId}: Document deleted`);
      return;
    }

    if (!before) {
      console.log(`[onRequestWrite] ${requestId}: Document created`);
      await handleNewRequest(requestId, after);
      return;
    }

    const statusChanged = before.status !== after.status;
    const mechanicCleared = before.mechanicId && !after.mechanicId;
    const declinedByGrew =
      after.status === 'requested' &&
      !after.mechanicId &&
      (after.declinedBy?.length ?? 0) > (before.declinedBy?.length ?? 0);

    if ((statusChanged && after.status === 'requested' && mechanicCleared) || declinedByGrew) {
      console.log(`[onRequestWrite] ${requestId}: Rematch triggered`, {
        statusChanged,
        mechanicCleared,
        declinedByGrew,
        beforeStatus: before.status,
        afterStatus: after.status,
        beforeDeclinedCount: before.declinedBy?.length ?? 0,
        afterDeclinedCount: after.declinedBy?.length ?? 0,
      });
      await handleRematch(requestId, after);
    }
  });

/** Core assignment: picks best available mechanic and writes transactionally. */
async function assignTopCandidate(
  requestId: string,
  request: ServiceRequest,
): Promise<boolean> {
  const excludeUids = request.declinedBy ?? [];
  const candidates = await findEligibleMechanics(db, request, excludeUids);
  if (candidates.length === 0) {
    console.log(`[assignTopCandidate] ${requestId}: No eligible mechanics found`);
    return false;
  }
  const top = candidates[0];
  const etaMin = deriveEtaMinutes(
    top.mechanic.baseLat && top.mechanic.baseLng
      ? { lat: top.mechanic.baseLat, lng: top.mechanic.baseLng }
      : undefined,
    request
  );
  const ref = db.collection('requests').doc(requestId);
  try {
    await db.runTransaction(async (tx) => {
      const snap = await tx.get(ref);
      if (!snap.exists) throw new Error('Request no longer exists');
      const cur = snap.data() as ServiceRequest;
      if (cur.status !== 'requested' || cur.mechanicId) throw new Error('already-assigned');
      tx.update(ref, {
        status: 'accepted',
        mechanicId: top.mechanic.id,
        mechanicName: top.mechanic.name,
        etaMin,
        updatedAt: new Date(),
      });
    });
    console.log(`[assignTopCandidate] ${requestId}: Assigned to ${top.mechanic.id} (${top.mechanic.name})`);
    return true;
  } catch (err: unknown) {
    const msg = err instanceof Error ? err.message : String(err);
    if (msg === 'already-assigned') {
      console.log(`[assignTopCandidate] ${requestId}: Skipped — already assigned by concurrent write`);
      return false;
    }
    throw err;
  }
}

/** Handle initial dispatch for a new request. */
async function handleNewRequest(requestId: string, request: ServiceRequest): Promise<void> {
  if (request.pickedMechanicId) {
    console.log(`[handleNewRequest] ${requestId}: Has pickedMechanicId, skipping auto-dispatch`);
    return;
  }
  if (request.status !== 'requested' || request.mechanicId) return;
  await assignTopCandidate(requestId, request);
}

/** Handle rematch when a job is released or declined. */
async function handleRematch(requestId: string, request: ServiceRequest): Promise<void> {
  if (request.status !== 'requested' || request.mechanicId) return;
  await assignTopCandidate(requestId, request);
}

/**
 * Scheduled function: runs every minute to find expired offers
 * and trigger rematch for them.
 */
export const scheduledOfferExpiry = functions.pubsub
  .schedule('every 1 minutes')
  .onRun(async () => {
    const now = Date.now();
    const snapshot = await db
      .collection('requests')
      .where('status', '==', 'requested')
      .where('offerExpiresAt', '<', now)
      .where('mechanicId', '==', null)
      .limit(50)
      .get();

    if (snapshot.empty) {
      console.log('[scheduledOfferExpiry] No expired offers to process');
      return;
    }

    console.log(`[scheduledOfferExpiry] Found ${snapshot.size} expired offers`);

    for (const doc of snapshot.docs) {
      const request = { id: doc.id, ...doc.data() } as ServiceRequest;
      if (!needsRematching(request)) continue;
      try {
        await handleRematch(request.id, request);
      } catch (err) {
        console.error('[scheduledOfferExpiry] ' + request.id + ' rematch failed:', err);
      }
    }
  });

/**
 * Scheduled function: runs every 5 minutes to clean up stale requested
 * offers that have no mechanic and no recent activity.
 * Marks them as cancelled after 30 minutes of no match.
 */
export const scheduledStaleRequestCleanup = functions.pubsub
  .schedule('every 5 minutes')
  .onRun(async () => {
    const staleThreshold = Date.now() - 30 * 60 * 1000;

    const snapshot = await db
      .collection('requests')
      .where('status', '==', 'requested')
      .where('mechanicId', '==', null)
      .where('updatedAt', '<', staleThreshold)
      .limit(20)
      .get();

    if (snapshot.empty) {
      return;
    }

    console.log(`[scheduledStaleRequestCleanup] Cancelling ${snapshot.size} stale requests`);

    const batch = db.batch();
    for (const doc of snapshot.docs) {
      const ref = doc.ref;
      batch.update(ref, {
        status: 'cancelled',
        cancellationReason: 'No mechanic available after 30 minutes',
        cancelledAt: new Date(),
        cancelledBy: 'system',
        updatedAt: new Date(),
      });
    }
    await batch.commit();
  });

/**
 * Callable function: manual trigger for admin to force rematch a specific request.
 */
export const forceRematch = functions.https.onCall(async (data, context) => {
  if (!context.auth?.token.admin) {
    throw new functions.https.HttpsError('permission-denied', 'Admin only');
  }

  const { requestId } = data as { requestId: string };
  if (!requestId) {
    throw new functions.https.HttpsError('invalid-argument', 'requestId required');
  }

  const doc = await db.collection('requests').doc(requestId).get();
  if (!doc.exists) {
    throw new functions.https.HttpsError('not-found', 'Request not found');
  }

  const request = { id: doc.id, ...doc.data() } as ServiceRequest;
  if (request.status !== 'requested' && !(request.status === 'accepted' || request.status === 'on-the-way' || request.status === 'arrived' || request.status === 'service')) {
    throw new functions.https.HttpsError('failed-precondition', 'Request is not in a rematchable state');
  }
  // Re-read inside transaction via assign path: reset then assign atomically where possible.
  // resetRequestForRematch does a single-field update; assignTopCandidate does a transaction with guard.
  await resetRequestForRematch(db, requestId);
  const freshSnap = await db.collection('requests').doc(requestId).get();
  const fresh = { ...(freshSnap.data() as ServiceRequest), id: freshSnap.id };
  await assignTopCandidate(requestId, fresh);

  return { success: true };
});

import { Firestore, Query, DocumentSnapshot, GeoPoint } from 'firebase-admin/firestore';
import { ServiceRequest, MechanicProfile } from './types';

export const OFFER_WINDOW_MS = 120_000;
export const MAX_CANDIDATES_PER_ROUND = 5;

/** Haversine distance in km between two lat/lng points. */
export function haversineKm(
  lat1: number,
  lng1: number,
  lat2: number,
  lng2: number
): number {
  const toRad = (d: number) => (d * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2;
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return 6371 * c;
}

/** Derive ETA in minutes from mechanic base to request location. */
export function deriveEtaMinutes(
  mechanicBase: { lat: number; lng: number } | undefined,
  request: ServiceRequest
): number {
  if (!mechanicBase) return request.etaMin ?? 15;
  const km = haversineKm(
    mechanicBase.lat,
    mechanicBase.lng,
    request.customerLat,
    request.customerLng
  );
  if (!Number.isFinite(km) || km <= 0) return request.etaMin ?? 15;
  return Math.max(1, Math.round((km / 35) * 60));
}

/** Score a mechanic for a request (higher = better match). */
export function scoreMechanic(
  mechanic: MechanicProfile,
  request: ServiceRequest
): number {
  if (!mechanic.approved || !mechanic.isAvailable) return -Infinity;
  if (!mechanic.baseLat || !mechanic.baseLng) return -Infinity;

  const km = haversineKm(
    mechanic.baseLat,
    mechanic.baseLng,
    request.customerLat,
    request.customerLng
  );
  if (!Number.isFinite(km)) return -Infinity;

  const radius = mechanic.serviceRadiusKm ?? 50;
  if (km > radius) return -Infinity;

  let score = 1000 - km * 10;

  if (request.problemCategory && mechanic.specialties?.includes(request.problemCategory)) {
    score += 200;
  }

  if (mechanic.rating != null) {
    score += mechanic.rating * 20;
  }

  if (mechanic.reviewCount != null) {
    score += Math.min(mechanic.reviewCount, 50) * 2;
  }

  if (mechanic.currentLat != null && mechanic.currentLng != null && mechanic.locationUpdatedAt) {
    const ageMs = Date.now() - mechanic.locationUpdatedAt;
    if (ageMs < 5 * 60 * 1000) {
      const liveKm = haversineKm(
        mechanic.currentLat,
        mechanic.currentLng,
        request.customerLat,
        request.customerLng
      );
      if (liveKm < km) {
        score += 100;
      }
    }
  }

  return score;
}

/** Find eligible mechanics for a request, sorted by score descending. */
export async function findEligibleMechanics(
  db: Firestore,
  request: ServiceRequest,
  excludeUids: string[] = []
): Promise<Array<{ mechanic: MechanicProfile; score: number }>> {
  const mechanicsRef = db.collection('mechanics');
  const snapshot = await mechanicsRef
    .where('approved', '==', true)
    .where('isAvailable', '==', true)
    .get();

  const candidates: Array<{ mechanic: MechanicProfile; score: number }> = [];

  for (const doc of snapshot.docs) {
    const mechanic = { id: doc.id, ...doc.data() } as MechanicProfile;
    if (excludeUids.includes(mechanic.id)) continue;

    const score = scoreMechanic(mechanic, request);
    if (score > -Infinity) {
      candidates.push({ mechanic, score });
    }
  }

  candidates.sort((a, b) => b.score - a.score);
  return candidates.slice(0, MAX_CANDIDATES_PER_ROUND);
}

/** Extend offer window for a request and clear assignment. Uses transaction so concurrent accepts don't race a reset. */
export async function resetRequestForRematch(
  db: Firestore,
  requestId: string
): Promise<boolean> {
  try {
    const ref = db.collection('requests').doc(requestId);
    await db.runTransaction(async (tx) => {
      const snap = await tx.get(ref);
      if (!snap.exists) throw new Error('Request no longer exists');
      tx.update(ref, {
        status: 'requested',
        mechanicId: null,
        mechanicName: null,
        mechanicLat: null,
        mechanicLng: null,
        offerExpiresAt: Date.now() + OFFER_WINDOW_MS,
        updatedAt: new Date(),
      });
    });
    return true;
  } catch (err) {
    console.error(`[resetRequestForRematch] ${requestId}:`, err);
    return false;
  }
}

/** Check if a request needs rematching (expired or released/declined). */
export function needsRematching(request: ServiceRequest): boolean {
  if (request.status !== 'requested') return false;
  if (request.mechanicId) return false;
  if (request.offerExpiresAt != null && request.offerExpiresAt < Date.now()) return true;
  return false;
}

#!/usr/bin/env node
import { strict as assert } from 'node:assert';
function memDb(seed) {
  const cols = new Map();
  for (const [name, docs] of Object.entries(seed || {})) {
    const m = new Map();
    for (const d of docs) m.set(d.id, { ...d });
    cols.set(name, m);
  }
  function col(name) {
    if (!cols.has(name)) cols.set(name, new Map());
    const m = cols.get(name);
    return {
      doc(id) {
        return {
          id,
          async get() {
            const data = m.get(id);
            return { exists: !!data, id, data: () => (data ? { ...data } : undefined), ref: this };
          },
          async update(patch) {
            const cur = m.get(id);
            if (!cur) throw new Error('not-found');
            m.set(id, { ...cur, ...patch });
          },
          async set(data) { m.set(id, { ...data, id }); },
        };
      },
      where(field, op, value) {
        const filters = [{ field, op, value }];
        const q = {
          where(f2, op2, v2) { filters.push({ field: f2, op: op2, value: v2 }); return q; },
          limit(n) { q._limit = n; return q; },
          async get() {
            let out = [...m.values()];
            for (const f of filters) {
              if (f.op === '==') out = out.filter((d) => d[f.field] === f.value);
              else if (f.op === '<') out = out.filter((d) => d[f.field] != null && d[f.field] < f.value);
              else throw new Error('unsupported op ' + f.op);
            }
            if (q._limit != null) out = out.slice(0, q._limit);
            const docs = out.map((d) => ({ id: d.id, data: () => ({ ...d }), ref: col(name).doc(d.id) }));
            return { empty: docs.length === 0, size: docs.length, docs };
          },
        };
        return q;
      },
      async get() {
        const docs = [...m.values()].map((d) => ({ id: d.id, data: () => ({ ...d }), ref: col(name).doc(d.id) }));
        return { empty: docs.length === 0, size: docs.length, docs };
      },
    };
  }
  return {
    collection: col,
    batch() {
      const ops = [];
      return { update(ref, patch) { ops.push([ref, patch]); }, async commit() { for (const [ref, patch] of ops) await ref.update(patch); } };
    },
    async runTransaction(fn) {
      const tx = { async get(ref) { return ref.get(); }, update(ref, patch) { return ref.update(patch); } };
      return fn(tx);
    },
  };
}

  });
  await check('forceRematch guard non-rematchable completed rejected', async () => {
    function isRematchable(s) { return s.status === 'requested' || ['accepted','on-the-way','arrived','service'].includes(s.status); }

  });

  });

#!/usr/bin/env node
/**
 * scripts/check-rules.mjs
 *
 * Static linter for `firestore.rules`. Java-free, no firebase-tools required.
 *
 * Why this exists
 * ---------------
 * The Firestore emulator-based linter (`firebase emulators:exec --only firestore`)
 * needs the JVM. This machine doesn't have one. The cached `firebase-tools`
 * builds in `~/.npm/_npx` no longer expose `lib/rules/rulesets` either, so the
 * offline `Ruleset` parser is no longer reachable.
 *
 * This script is the substitute: a small set of targeted checks against the
 * rules text that catch the specific Phase 9 invariants we care about. It is
 * NOT a full grammar parse — it only looks for the lines/blocks we want to
 * be present.
 *
 * Exit codes: 0 if all checks pass, 1 if any fail. Failing checks print
 * `FAIL:` lines so CI/log scrapers can grep for them.
 *
 * Usage
 * -----
 *   node scripts/check-rules.mjs                 # lints firestore.rules in repo root
 *   node scripts/check-rules.mjs path/to/rules   # lints an arbitrary file
 *
 * Checks
 * ------
 *  R001  isClaimAdmin() helper checks request.auth.token.admin == true
 *  R002  users/{uid} create rule restricts role to ['customer','mechanic']
 *  R003  users/{uid} update rule forbids mutating 'approved'
 *  R004  requests/{id} read allows customer + mechanic-queue + admin
 *  R005  requests/{id} create requires status == 'requested'
 *  R006  requests/{id} update has the customer-cancel/rate branch AND its
 *        hasOnly([...]) permits cancelledAt / cancelledBy / cancellationReason
 *  R007  mechanic-advance branch freezes mechanicId/mechanicName
 *  R008  requests/{id} update has the FIRST-ASSIGNMENT branch
 *  R009  FIRST-ASSIGNMENT branch enforces request.time < offerExpiresAt
 *  R010  FIRST-ASSIGNMENT branch restricts affected keys to
 *        {status, mechanicId, mechanicName, etaMin, updatedAt}
 *  R011  mechanics/{uid} create forces approved == false
 *  R012  mechanics/{uid} self-update forbids rating/reviewCount mutation
 *  R013  catch-all {document=**} denies everything
 */

import { readFileSync, existsSync } from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = dirname(fileURLToPath(import.meta.url));

  process.stderr.write(`ERROR: rules file not found: ${filePath}\n`);
  process.exit(2);

const src = readFileSync(filePath, 'utf8');

function matchBlock(pathPattern) {
  // The block header in this repo is `match /<pattern> {` — note there is
  // NO closing slash between the path and the block-opening brace, even
  // for the top-level `match /databases/{database}/documents {`. We use a
  // regex anchored on the path itself plus the literal trailing ` {`.
  // We escape every regex metachar so callers can pass patterns like
  // `{document=**}` without thinking about it.
  const escaped = pathPattern.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const re = new RegExp(`match\\s+/${escaped}\\s*\\{`);
  const m = re.exec(src);
  if (!m) return null;
  const open = src.indexOf('{', m.index + m[0].length - 1);
  if (open < 0) return null;
  let depth = 1;
  let i = open + 1;
  while (i < src.length && depth > 0) {
    const ch = src[i];
    if (ch === '{') depth++;
    else if (ch === '}') depth--;
    i++;
  }
  return src.slice(open, i);
}

function hasHelper(name) {
  return new RegExp(`function\\s+${name}\\s*\\(`).test(src);
}

function helperBody(name) {
  const re = new RegExp(`function\\s+${name}\\s*\\([^)]*\\)\\s*\\{([\\s\\S]*?)\\n\\s*\\}`, 'm');
  const m = src.match(re);
  return m ? m[1] : null;
}

function record(id, label, pass, detail) {
  results.push({ id, label, pass, detail });
}

/**
 * Mechanic-facing data access. All accessors are null-safe: if Firebase
 * is not configured, they return early and the page renders the empty
 * or error state itself.
 */
import {
  collection, doc, getDoc, getDocs, limit, onSnapshot, orderBy, query, runTransaction,
  serverTimestamp, updateDoc, where,
  type DocumentData, type Unsubscribe,
} from 'firebase/firestore';
import { db } from '@/app/firebase';
import type { ServiceRequest } from './customer';
import { OFFER_WINDOW_MS } from './customer';

function requireDb() {
  const handle = db();
  if (!handle) throw new Error('Firebase is not configured.');
  return handle;
}

function asRequest(id: string, raw: DocumentData | undefined): ServiceRequest | null {
  if (!raw) return null;
  return { id, ...(raw as Omit<ServiceRequest, 'id'>) };
}

export type JobStatus = ServiceRequest['status'];

export interface MechanicProfile {

  email?: string;
  /** Verification lifecycle for a newly-registered mechanic. */
  verificationStatus?: 'pending' | 'under_review' | 'approved' | 'rejected';
  /** True when an admin has approved this mechanic (`verificationStatus === 'approved'`). */
  approved: boolean;
  specialties: string[];
  /** Supported vehicle types (e.g. ['car','motorcycle','suv']). */
  vehicleTypes?: string[];
  /** Years of professional experience. */
  experienceYears?: number;
  /** Willing to travel up to this many km from the base. */
  serviceRadiusKm?: number;
  isAvailable: boolean;
  rating?: number;
  reviewCount?: number;
  vehicleDescription?: string;
  /** Workshop name (or mobile rig name). */
  garageName?: string;

  garageAddress?: string;

  currentLat?: number;
  /** Current (live) coordinates from watchPosition. */
  currentLng?: number;

  locationUpdatedAt?: number;
  /** Base (home/garage) location — used for "nearby" distance and ETA. */
  baseLat?: number;
  baseLng?: number;
}

/**
 * Live queue of jobs in 'requested' state, ordered by recency.
 *
 * Firestore has no `array-not-contains` / `array-does-not-contain`,
 * so a server-side `where('declinedBy', 'not-contains', uid)` is
 * not possible. The matching engine already excludes `declinedBy`
 * on the server (`findEligibleMechanics(..., excludeUids)`), but
 * the dispatch queue snapshot would still *deliver* the declined
 * doc to the mechanic's listener. We hide it client-side here
 * instead — no new composite index and no rules change required.
 *
 * Pass the current mechanic's uid as `excludeUid` so they never
 * see a request they already declined. Callers that omit it keep
 * the previous behaviour (no filtering).
 */
export function watchIncoming(
  cb: (jobs: ServiceRequest[]) => void,
  excludeUid?: string,
): Unsubscribe | null {
  try {
    const d = requireDb();
    return onSnapshot(
      query(collection(d, 'requests'), where('status', '==', 'requested'), orderBy('createdAt', 'desc'), limit(20)),
      (snap) => {
        const jobs = snap.docs
          .map((s) => asRequest(s.id, s.data()))
          .filter((j): j is ServiceRequest => Boolean(j))
          .filter((j) => !excludeUid || !(j.declinedBy ?? []).includes(excludeUid));
        cb(jobs);
      },
      () => cb([]),
    );
  } catch {
    return null;
  }
}

/** Tri-state result of an accept attempt. */
export type AcceptResult = 'accepted' | 'expired' | 'alreadyTaken' | 'error';

/** Pick a sensible ETA when the customer didn't pass one in. */
function deriveEta(
  mechanicBase: { lat: number; lng: number } | undefined,
  request: ServiceRequest,
): number {
  if (mechanicBase) {
    const toRad = (d: number) => (d * Math.PI) / 180;
    const dLat = toRad(request.customerLat - mechanicBase.lat);
    const dLng = toRad(request.customerLng - mechanicBase.lng);
    const lat1 = toRad(mechanicBase.lat);
    const lat2 = toRad(request.customerLat);
    const h = Math.sin(dLat / 2) ** 2 + Math.cos(lat1) * Math.cos(lat2) * Math.sin(dLng / 2) ** 2;
    const km = 2 * 6371 * Math.atan2(Math.sqrt(h), Math.sqrt(1 - h));
    if (Number.isFinite(km) && km > 0) {
      return Math.max(1, Math.round((km / 35) * 60));
    }
  }
  return request.etaMin ?? 15;
}

/**
 * Accept a queued request.
 *
 * Uses a Firestore transaction so the read-then-write is atomic:
 *   - read the request doc
 *   - if `status !== 'requested'`, return `alreadyTaken`
 *   - if `offerExpiresAt < now`, return `expired`
 *   - otherwise write `accepted` + mechanicId/mechanicName/etaMin/updatedAt
 *     and return `accepted`
 *
 * Pages that only need the boolean ("did it work?") can use:
 *   `result === 'accepted'`.
 */
export async function acceptJob(
  id: string,
  mechanicId: string,
  mechanicName: string,
  etaMin?: number,
  mechanicBase?: { lat: number; lng: number },
): Promise<AcceptResult> {
  try {
    const d = requireDb();
    const ref = doc(d, 'requests', id);
    const result = await runTransaction(d, async (tx) => {
      const snap = await tx.get(ref);
      if (!snap.exists()) return 'alreadyTaken' as const;
      const data = snap.data() as ServiceRequest;
      if (data.status !== 'requested') return 'alreadyTaken' as const;
      if (data.offerExpiresAt != null && data.offerExpiresAt < Date.now()) {
        return 'expired' as const;
      }
      const eta = etaMin ?? deriveEta(mechanicBase, data);
      tx.update(ref, {
        status: 'accepted',
        mechanicId,
        mechanicName,
        etaMin: eta,
        updatedAt: serverTimestamp(),
      });
      return 'accepted' as const;
    });
    return result;
  } catch {
    return 'error';
  }
}

export async function updateJobStatus(id: string, status: JobStatus): Promise<boolean> {
  try {
    const d = requireDb();
    await updateDoc(doc(d, 'requests', id), { status, updatedAt: serverTimestamp() });
    return true;
  } catch { return false; }
}

/**
 * Decline an offer while the request is still in 'requested' state.
 *
 * We don't change the request status — only append the mechanic's uid
 * to `declinedBy[]`. The offer window keeps ticking for the remaining
 * candidates. Operators see the declinedBy list as a signal that a
 * mechanic is cycling through offers without accepting.
 *
 * A second decline from the same mechanic is a no-op (uid is deduped
 * in the transaction).
 */
export async function declineOffer(
  requestId: string,
  mechanicId: string,
): Promise<'declined' | 'alreadyTaken' | 'forbidden' | 'error'> {
  try {
    const d = requireDb();
    const ref = doc(d, 'requests', requestId);
    return await runTransaction(d, async (tx) => {
      const snap = await tx.get(ref);
      if (!snap.exists()) return 'alreadyTaken' as const;
      const data = snap.data() as ServiceRequest;
      if (data.status !== 'requested') return 'alreadyTaken' as const;
      const list = Array.isArray(data.declinedBy) ? data.declinedBy : [];
      if (list.includes(mechanicId)) return 'declined' as const;
      tx.update(ref, {
        declinedBy: [...list, mechanicId],
        updatedAt: serverTimestamp(),
      });
      return 'declined' as const;
    });
  } catch (e: unknown) {
    const code = (e as { code?: string } | null)?.code;
    if (code === 'permission-denied') return 'forbidden';
    return 'error';
  }
}

/** Statuses where the mechanic can still release a job back to the queue. */
const RELEASABLE_STATUSES = ['accepted', 'on-the-way', 'arrived', 'service'] as const;

/**
 * Release an accepted job back to the queue.
 *
 *  - status  -> 'requested'
 *  - mechanicId / mechanicName cleared so the offer is open again
 *  - offerExpiresAt extended by OFFER_WINDOW_MS so the next candidate
 *    gets a fresh window
 *  - releaseReason stored for operator audit
 *
 * The Firestore rule that protects this branch must use hasOnly(...)
 * so the mechanic cannot write arbitrary fields.
 */
export async function releaseJob(
  requestId: string,
  mechanicId: string,
  reason: string,
): Promise<'released' | 'not_assigned' | 'invalid_state' | 'forbidden' | 'error'> {
  try {
    const d = requireDb();
    const ref = doc(d, 'requests', requestId);
    return await runTransaction(d, async (tx) => {
      const snap = await tx.get(ref);
      if (!snap.exists()) return 'invalid_state' as const;
      const data = snap.data() as ServiceRequest;
      if (data.mechanicId !== mechanicId) return 'not_assigned' as const;
      if (!(RELEASABLE_STATUSES as readonly string[]).includes(data.status)) {
        return 'invalid_state' as const;
      }
      tx.update(ref, {
        status: 'requested',
        mechanicId: null,
        mechanicName: null,
        mechanicLat: null,
        mechanicLng: null,
        offerExpiresAt: Date.now() + OFFER_WINDOW_MS,
        releaseReason: reason,
        releasedAt: serverTimestamp(),
        updatedAt: serverTimestamp(),
      });
      return 'released' as const;
    });
  } catch (e: unknown) {
    const code = (e as { code?: string } | null)?.code;
    if (code === 'permission-denied') return 'forbidden';
    return 'error';
  }
}

    const snap = await getDocs(
      query(collection(d, 'requests'),
        where('mechanicId', '==', mechanicId),
        where('status', 'in', ['accepted', 'on-the-way', 'arrived', 'service']),
        limit(1),
      ),
    );
    if (snap.empty) return null;
    const doc0 = snap.docs[0];
    return asRequest(doc0.id, doc0.data());
  } catch { return null; }
}
